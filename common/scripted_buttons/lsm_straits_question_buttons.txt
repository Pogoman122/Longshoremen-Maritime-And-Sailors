# AI voting is handled by one weighted draw in the journal, not independent button rolls.

straits_ancient_rule_button = {
    name = straits_ancient_rule_button
    desc = straits_ancient_rule_button_desc
    visible = {
        has_journal_entry = je_straits_question
        lsm_straits_is_voter = yes
        NOT = { has_global_variable = straits_conference_has_concluded_global_var }
    }
    selected = { has_variable = straits_ancient_rule_button_var }
    possible = {
        custom_tooltip = {
            text = lsm_straits_voting_open_tt
            global_var:je_straits_question_timer > 0
        }
        custom_tooltip = {
            text = lsm_straits_not_voted_tt
            lsm_straits_has_voted = no
        }
    }
    ai_chance = { value = 0 }
    effect = {
        custom_tooltip = {
            text = lsm_straits_vote_ancient_rule_tt
            lsm_straits_cast_ancient_rule_vote = yes
        }
    }
}
