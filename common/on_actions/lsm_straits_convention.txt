on_monthly_pulse_country = {
    on_actions = { lsm_straits_question_monthly_pulse }
}
lsm_straits_question_monthly_pulse = {
    trigger = {
        lsm_straits_is_chair = yes
        exists = c:TUR
        OR = {
            game_date >= 1841.7.1
            has_global_variable = hunkar_iskelesi_expired_global_var
        }
        NOT = { has_global_variable = straits_conference_started_global_var }
    }
    effect = { trigger_event = { id = straits_question_events.1 popup = yes } }
}

on_country_withdrawn_from_treaty = {
    on_actions = { lsm_hunkar_iskelesi_withdrawal }
}
on_country_broke_treaty = {
    on_actions = { lsm_hunkar_iskelesi_withdrawal }
}
lsm_hunkar_iskelesi_withdrawal = {
    trigger = {
        any_scope_article = { has_type = hunkar_iskelesi }
        NOT = { has_global_variable = straits_conference_started_global_var }
    }
    effect = {
        every_country = {
            limit = { lsm_straits_is_chair = yes }
            trigger_event = { id = straits_question_events.1 days = 1 popup = yes }
        }
    }
}
